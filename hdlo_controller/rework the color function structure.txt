 Colorfunctions have basically one purpose: to return a CRGB given a float position and possibly some other parameters; they are always called by a string name, like "fire2012" from within a sequence in sequence.cpp, which has these names, plus palette names, and other information. 


Rework the colorfunction structure into a proper class, which encompasses both static and dynamic color functions; colorfunctions will now  override to take more kinds of vairables.

1. Color Function Class Instead of Function Pointers


cpp
// Base class for all color functions
class IColorFunction {
public:
    virtual ~IColorFunction() {}
    virtual CRGB getColor(float position) = 0;
    virtual void setParameters(const std::vector<FunctionParameter>& params) {}
    virtual IColorFunction* clone() = 0;  // For copying
};

// Example implementation

class PlasmaFunction : public IColorFunction {
private:
    float intensity = 1.0f;
    float speed = 7.0f;
    
public:
    CRGB getColor(float position) override {
        uint8_t hue = beatsin8(speed * intensity, 0, 255) + (position * 255.0f);
        uint8_t brightness = constrain(255 * intensity, 0, 255);
        return CHSV(hue, 255, brightness);
    }
    
    void setParameters(const std::vector<FunctionParameter>& params) override {
        if(params.size() > 0 && params[0].type == FunctionParameter::FLOAT) {
            intensity = constrain(params[0].floatVal, 0.1f, 2.0f);
        }
        if(params.size() > 1 && params[1].type == FunctionParameter::FLOAT) {
            speed = constrain(params[1].floatVal, 1.0f, 20.0f);
        }
    }
    
    IColorFunction* clone() override {
        return new PlasmaFunction(*this);
    }
};
2. Function Factory Pattern


cpp
class ColorFunctionFactory {
private:
    std::map<String, std::function<IColorFunction*()>> creators;
    
public:
    void registerFunction(const String& name, std::function<IColorFunction*()> creator) {
        creators[name] = creator;
    }
    
    IColorFunction* create(const String& name) {
        auto it = creators.find(name);
        if(it != creators.end()) {
            return it->second();
        }
        return nullptr;
    }
    
    // Initialize with all functions
    void initialize() {
        registerFunction("plasma", []() { return new PlasmaFunction(); });
        registerFunction("fire", []() { return new FireFunction(); });
        registerFunction("rainbow", []() { return new RainbowFunction(); });
        // etc...
    }
};

3. Model Stores Function Instances


cpp
class colormodel {
private:
    std::array<std::unique_ptr<IColorFunction>, 120> edgeFunctions;
    std::array<String, 120> edgePalettes;
    
public:
    void setEdgeFunction(int edge, const String& funcName, 
                         const std::vector<FunctionParameter>& params,
                         const String& palette) {
        // Create function instance
        edgeFunctions[edge].reset(functionFactory.create(funcName));
        
        if(edgeFunctions[edge]) {
            // Apply parameters
            edgeFunctions[edge]->setParameters(params);
            
            // Store palette
            edgePalettes[edge] = palette;
        }
    }
    
    CRGB getColor(int edge, float position) {
        if(edgeFunctions[edge]) {
            // Apply palette if needed
            if(!edgePalettes[edge].empty()) {
                setCurrentPalette(edgePalettes[edge]);
            }
            return edgeFunctions[edge]->getColor(position);
        }
        return CRGB::Black;
    }
};

class modelsequence {
private:
    std::unique_ptr<colormodel> currentModel;
    
    void applyStep(const SequenceStep& step) {
        currentModel = createModel(step.modelName);
        
        // Apply functions based on model's edge mapping
        for(int edge = 0; edge < 120; edge++) {
            int funcIndex = getModelFunctionIndex(step.modelName, edge);
            
            if(funcIndex < step.functions.size()) {
                const FunctionSpec& spec = step.functions[funcIndex];
                currentModel->setEdgeFunction(edge, 
                                             spec.functionName,
                                             spec.parameters,
                                             spec.paletteName);
            }
        }
    }
    
public:
    CRGB getColor(int edge, float position) {
        if(currentModel) {
            return currentModel->getColor(edge, position);
        }
        return CRGB::Black;
    }
};
